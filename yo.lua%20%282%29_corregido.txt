    local function getHWID()

    local function generateHiddenDots(availableWidth, charWidth)

    local function formatTime12()
        local hour = tonumber(os.date("%H"))
        local min = os.date("%M")
        local sec = os.date("%S")
        local period = "AM"
        if hour >= 12 then period = "PM" end
        if hour > 12 then hour = hour - 12 end
        if hour == 0 then hour = 12 end
        return string.format("%d:%s:%s %s", hour, min, sec, period)
    end

    local function formatDate()
        return os.date("%b %d, %Y")
    end

    local function setupDragging(header, main)
        if not Options.Draggable then return end
        local dragging, dragStart, startPos, dragInput
        header.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = input.Position
                startPos = main.Position
                dragInput = input
                input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then
                        if dragInput == input then dragging = false dragInput = nil end
                    end
                end)
            end
        end)
        UserInputService.InputChanged:Connect(function(input)
            if not dragging or not dragInput then return end
            if input.UserInputType == Enum.UserInputType.MouseMovement then
                local delta = input.Position - dragStart
                main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            elseif input.UserInputType == Enum.UserInputType.Touch then
                if input == dragInput then
                    local delta = input.Position - dragStart
                    main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
                end
            end
        end)
    end

        logoImage.ImageColor3 = Theme.Text
        logoImage.ScaleType = Enum.ScaleType.Fit
        logoImage.ZIndex = 54
        logoImage.Parent = overlay

        local function openDoors(callback)
            TweenService:Create(logoImage, TweenInfo.new(0.2, Enum.EasingStyle.Quart), {ImageTransparency = 1}):Play()
            task.wait(0.25)
            TweenService:Create(leftDoor, TweenInfo.new(0.4, Enum.EasingStyle.Quart, Enum.EasingDirection.In), {Position = UDim2.new(0, -halfWidth, 0, 0)}):Play()

            overlay.Visible = true

            task.wait(0.3)
            if callback then callback() end
        end

    local function BuildCenteredUI(windowWidth, windowHeight, panelHeight, userPanelWidth, changelogPanelWidth, gap, buildContent)
        local gui = buildContent.gui

        local container = Instance.new("Frame")
        container.Name = "Container"
        container.Size = UDim2.new(0, windowWidth, 0, panelHeight)
        container.Position = UDim2.new(0.5, 0, 1.5, 0)
        container.AnchorPoint = Vector2.new(0.5, 0.5)
        container.BackgroundTransparency = 1
        container.Parent = gui

        local mainFrame = Instance.new("Frame")
        mainFrame.Name = "MainFrame"
        mainFrame.Size = UDim2.new(0, windowWidth, 0, windowHeight)
        mainFrame.Position = UDim2.new(0.5, 0, 0, 0)
        mainFrame.AnchorPoint = Vector2.new(0.5, 0)
        mainFrame.BackgroundColor3 = Theme.Background
        mainFrame.BorderSizePixel = 0
        mainFrame.Parent = container
        Instance.new("UICorner", mainFrame).CornerRadius = UDim.new(0, 4)

        local mainStroke = Instance.new("UIStroke", mainFrame)
        mainStroke.Color = Theme.Accent
        mainStroke.Thickness = 2
        mainStroke.Transparency = 0.4

        local userPanel, toggleUserPanel, isUserOpen, userPanelActualWidth = CreateUserInfoPanel(container, windowWidth, panelHeight, userPanelWidth, mainFrame, gap, false)
        local changelogPanel, toggleChangelog, isChangelogOpen, changelogPanelActualWidth = CreateChangelogPanel(container, windowWidth, panelHeight, changelogPanelWidth, mainFrame, gap)

        local function getContainerWidth()
            local w = windowWidth
            if isUserOpen() then w = w + gap + userPanelActualWidth end
            if isChangelogOpen() then w = w + gap + changelogPanelActualWidth end
            return w
        end

        local function toggleUser(userIcon)
            local currentW = getContainerWidth()
            if isUserOpen() then
                toggleUserPanel(userIcon, container, currentW - gap - userPanelActualWidth)
            else
                toggleUserPanel(userIcon, container, currentW)
            end
        end

        local function toggleCL(changelogIcon)
            local currentW = getContainerWidth()
            if isChangelogOpen() then
                toggleChangelog(changelogIcon, container, currentW - gap - changelogPanelActualWidth)
            else
                toggleChangelog(changelogIcon, container, currentW)
            end
        end

        local function closeAllPanels(userIcon, changelogIcon, callback)
            if isChangelogOpen() then toggleCL(changelogIcon) task.wait(0.35) end
            if isUserOpen() then toggleUser(userIcon) task.wait(0.35) end
            if callback then callback() end
        end

        return {
            container = container,
            mainFrame = mainFrame,
            mainStroke = mainStroke,
            toggleUser = toggleUser,
            toggleCL = toggleCL,
            isUserOpen = isUserOpen,
            isChangelogOpen = isChangelogOpen,
            closeAllPanels = closeAllPanels
        }
    end

    local function refreshTheme(folder)
        local newT = buildTheme()
        folder = folder or guiRoot
        if not folder then
            return
        end
        local mapping = {
            { old = prevTheme.Background, new = newT.Background },
            { old = prevTheme.Header, new = newT.Header },
            { old = prevTheme.Sidebar, new = newT.Sidebar },
            { old = prevTheme.CardBG, new = newT.CardBG },
            { old = prevTheme.CardStroke, new = newT.CardStroke },
            { old = prevTheme.Accent, new = newT.Accent },
            { old = prevTheme.TextWhite, new = newT.TextWhite },
            { old = prevTheme.TextMuted, new = newT.TextMuted },
            { old = prevTheme.RedText, new = newT.RedText },
            { old = prevTheme.Control, new = newT.Control },
            { old = prevTheme.ControlAlt, new = newT.ControlAlt },
            { old = prevTheme.ControlRaised, new = newT.ControlRaised },
            { old = prevTheme.Hover, new = newT.Hover },
            { old = prevTheme.Success, new = newT.Success },
        }
        if Theme then
            table.insert(mapping, { old = Theme.Accent, new = newT.Accent })
        end
        local props = {
            "BackgroundColor3",
            "TextColor3",
            "ImageColor3",
            "BorderColor3",
            "ScrollBarImageColor3",
        }
        for _, child in ipairs(folder:GetDescendants()) do
            for _, prop in ipairs(props) do
                local cur = child[prop]
                if typeof(cur) == "Color3" then
                    for _, m in ipairs(mapping) do
                        if cur == m.old then
                            child[prop] = m.new
                            break
                        end
                    end
                end
            end
            if child:IsA("UIStroke") then
                local cur = child.Color
                if typeof(cur) == "Color3" then
                    for _, m in ipairs(mapping) do
                        if cur == m.old then
                            child.Color = m.new
                            break
                        end
                    end
                end
            end
        end
        prevTheme = {
            Background = newT.Background,
            Sidebar = newT.Sidebar,
            CardBG = newT.CardBG,
            CardStroke = newT.CardStroke,
            Accent = newT.Accent,
            TextWhite = newT.TextWhite,
            TextMuted = newT.TextMuted,
            RedText = newT.RedText,
            Control = newT.Control,
            ControlAlt = newT.ControlAlt,
            ControlRaised = newT.ControlRaised,
            Hover = newT.Hover,
        }
        Theme = newT
        local mainScale = MainFrame and MainFrame:FindFirstChildOfClass("UIScale")
        if mainScale then
            mainScale.Scale = compactScale()
        end
    end

    local fadeStore = setmetatable({}, { __mode = "k" })
    local function storeFadeValue(inst, prop)
        local entry = fadeStore[inst]
        if not entry then
            entry = {}
            fadeStore[inst] = entry
        end
        if entry[prop] == nil then
            entry[prop] = inst[prop]
        end
    end

    local function collectFadeProps(inst)
        local props = {}
        if inst:IsA("Frame") or inst:IsA("ScrollingFrame") then
            table.insert(props, "BackgroundTransparency")
        end
        if inst:IsA("TextLabel") or inst:IsA("TextButton") then
            table.insert(props, "TextTransparency")
        end
        if inst:IsA("ImageLabel") then
            table.insert(props, "ImageTransparency")
        end
        if inst:IsA("UIStroke") then
            table.insert(props, "Transparency")
        end
        if inst:IsA("CanvasGroup") then
            table.insert(props, "GroupTransparency")
        end
        return props
    end

    local function fadeGroup(root, target, store, duration)
        local items = { root }
        for _, c in ipairs(root:GetDescendants()) do
            table.insert(items, c)
        end
        for _, g in ipairs(items) do
            for _, prop in ipairs(collectFadeProps(g)) do
                if store then
                    storeFadeValue(g, prop)
                end
                TweenService:Create(g, TweenInfo.new(duration or 0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { [prop] = target }):Play()
            end
        end
    end

    local function restoreFade(root)
        local items = { root }
        for _, c in ipairs(root:GetDescendants()) do
            table.insert(items, c)
        end
        for _, g in ipairs(items) do
            local saved = fadeStore[g]
            if saved then
                for prop, value in pairs(saved) do
                    TweenService:Create(g, TweenInfo.new(0.24, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { [prop] = value }):Play()
                end
            end
        end
    end

        local _cachedPrefFootRef = nil

        local function getPreferredFoot()
            local preferredFoot = _cachedPrefFootRef

            if not (preferredFoot and preferredFoot.Parent) then
                local folder = Lighting and Lighting:FindFirstChild(LocalPlayer.Name)
                preferredFoot = folder and folder:FindFirstChild("PreferredFoot")
                _cachedPrefFootRef = preferredFoot
            end

            if preferredFoot then
                local ok, value = pcall(function()
                    return preferredFoot.Value
                end)

                if ok and (value == 1 or value == 2) then
                    return value
                end
            end

            return 1
        end

        local function getKickPart(character)
            if not character then
                return nil
            end

            local humanoid = character:FindFirstChildOfClass("Humanoid")
            local preferredFoot = getPreferredFoot()

            if humanoid and humanoid.RigType == Enum.HumanoidRigType.R15 then
                if preferredFoot == 2 then
                    return character:FindFirstChild("LeftLowerLeg")
                        or character:FindFirstChild("LeftUpperLeg")
                end

                return character:FindFirstChild("RightLowerLeg")
                    or character:FindFirstChild("RightUpperLeg")
            end

            if preferredFoot == 2 then
                return character:FindFirstChild("Left Leg")
            end

            return character:FindFirstChild("Right Leg")
        end

        local function performTPSContact(kickPart, tpsBall)
            if not tpsBall or not tpsBall.Parent then
                return false
            end

            if not firetouchinterest then
                return false
            end

            local character = getCharacter()
            if not character then
                return false
            end

            local root = character:FindFirstChild("HumanoidRootPart")
            local rleg = character:FindFirstChild("RightLowerLeg")
                or character:FindFirstChild("Right Leg")
            local lleg = character:FindFirstChild("LeftLowerLeg")
                or character:FindFirstChild("Left Leg")

            if root then
                firetouchinterest(tpsBall, root, 0)
                firetouchinterest(tpsBall, root, 1)
            end

            if rleg then
                firetouchinterest(tpsBall, rleg, 0)
                firetouchinterest(tpsBall, rleg, 1)
            end

            if lleg then
                firetouchinterest(tpsBall, lleg, 0)
                firetouchinterest(tpsBall, lleg, 1)
            end

            return true
        end

            local HttpService = game:GetService("HttpService")
            local TeleportService = game:GetService("TeleportService")

            local cursor = ""

            repeat
                local page
                local ok = pcall(function()
                    local url =
                        "https://games.roblox.com/v1/games/"
                        .. tostring(game.PlaceId)
                        .. "/servers/Public?sortOrder="
                        .. tostring(sortOrder)
                        .. "&limit=100"
                        .. (cursor ~= "" and ("&cursor=" .. cursor) or "")

                end)

                if ok and page and page.data then
                    for _, server in ipairs(page.data) do
                        if server.id ~= game.JobId and server.playing < server.maxPlayers then
                            table.insert(servers, server)
                        end
                    end

                    cursor = page.nextPageCursor or ""
                else
                    cursor = ""
                end
            until cursor == ""

                local server = servers[math.random(1, math.min(3, #servers))]
                TeleportService:TeleportToPlaceInstance(
                    game.PlaceId,
                    server.id,
                    player

        Callbacks.ImpactDistance = function(enabled)
            if enabled then
                v8ImpactFolder = Instance.new("Folder")
                v8ImpactFolder.Name = "ImpactDistanceFX"
                v8ImpactFolder.Parent = workspace

                v8ImpactMarker = Instance.new("Part")
                v8ImpactMarker.Shape = Enum.PartType.Cylinder
                v8ImpactMarker.Size = Vector3.new(0.1, 4, 4)
                v8ImpactMarker.Color = Color3.fromRGB(255, 76, 76)
                v8ImpactMarker.Material = Enum.Material.Neon
                v8ImpactMarker.Anchored = true
                v8ImpactMarker.CanCollide = false
                v8ImpactMarker.Transparency = 0.4
                v8ImpactMarker.CFrame =
                    CFrame.new()
                    * CFrame.Angles(
                        0,
                        0,
                        math.rad(90)
                    )
                v8ImpactMarker.Parent = v8ImpactFolder

                v8ImpactConnection =
                    V8RunService.RenderStepped:Connect(function()
                        local ball = v8NearestFootball()

                        if not ball
                            or ball.AssemblyLinearVelocity.Magnitude <= 0.5 then
                            v8ImpactFolder.Parent = nil
                            return
                        end

                        v8ImpactFolder.Parent = workspace

                        local position = ball.Position
                        local velocity = ball.AssemblyLinearVelocity
                        local gravity = Vector3.new(
                            0,
                            -workspace.Gravity,
                            0
                        )

                        local raycastParams = RaycastParams.new()
                        raycastParams.FilterType =
                            Enum.RaycastFilterType.Exclude

                        
                        
                        raycastParams.FilterDescendantsInstances = {
                            ball,
                            v8ImpactFolder,
                        }

                        local impactPosition

                        for t = 0, 3, 0.05 do
                            local p0 =
                                position
                                + velocity * t
                                + (0.5 * gravity) * (t ^ 2)

                            local nextT = t + 0.05

                            local p1 =
                                position
                                + velocity * nextT
                                + (0.5 * gravity) * (nextT ^ 2)

                            local result = workspace:Raycast(
                                p0,
                                p1 - p0,
                                raycastParams
                            )

                            if result then
                                impactPosition = result.Position
                                break
                            end
                        end

                        if impactPosition then
                            v8ImpactMarker.Transparency = 0.4
                            v8ImpactMarker.CFrame =
                                CFrame.new(impactPosition)
                                * CFrame.Angles(
                                    0,
                                    0,
                                    math.rad(90)
                                )
                        else
                            v8ImpactMarker.Transparency = 1
                        end
                    end)

                return
            end

            if v8ImpactConnection then
                v8ImpactConnection:Disconnect()
                v8ImpactConnection = nil
            end

            if v8ImpactFolder then
                v8ImpactFolder:Destroy()
                v8ImpactFolder = nil
            end

            v8ImpactMarker = nil
        end

        local function v8CopyAvatar(userId, character)
            if not character then
                return
            end

            local humanoid =
                character:FindFirstChildOfClass("Humanoid")

            if not humanoid then
                return
            end

            v8StopAnimations(humanoid)

            local appearanceOK, appearanceModel = pcall(function()
                return V8Players:GetCharacterAppearanceAsync(userId)
            end)

            local descriptionOK, description = pcall(function()
                return V8Players:GetHumanoidDescriptionFromUserId(userId)
            end)

            
            
            if not appearanceOK or not appearanceModel then
                if descriptionOK and description then
                    v8ApplyDescription(humanoid, description)
                    v8StopAnimations(humanoid)
                end

                return
            end

            for _, object in pairs(character:GetChildren()) do
                if object:IsA("Accessory")
                    or object:IsA("Shirt")
                    or object:IsA("Pants")
                    or object:IsA("CharacterMesh")
                    or object:IsA("BodyColors") then
                    object:Destroy()
                end
            end

            local targetHead =
                character:FindFirstChild("Head")

            if targetHead then
                for _, object in pairs(targetHead:GetChildren()) do
                    if object:IsA("Decal") then
                        object:Destroy()
                    end
                end
            end

            for _, object in ipairs(
                appearanceModel:GetDescendants()
            ) do
                pcall(function()
                    v8CopyAppearanceObject(
                        object,
                        character,
                        humanoid
                    )
                end)
            end

            local faceTexture

            for _, object in ipairs(
                appearanceModel:GetDescendants()
            ) do
                if object:IsA("Decal")
                    and (
                        object.Name == "face"
                        or object.Name == "Face"
                    ) then
                    faceTexture = object.Texture
                end
            end

            if targetHead then
                local face = Instance.new("Decal")
                face.Name = "face"
                face.Face = Enum.NormalId.Front
                face.Texture =
                    faceTexture
                    or "rbxasset://textures/face.png"
                face.Parent = targetHead
            end

            if descriptionOK and description then
                v8ApplyDescription(humanoid, description)
            end

            local oldParent = character.Parent
            character.Parent = nil
            character.Parent = oldParent

            v8StopAnimations(humanoid)

            appearanceModel:Destroy()
        end

                local velMag = bv.Magnitude
                lead = lead + math.min(math.max(velMag / 70, 0), 0.30)
                local dt = now - _reactAlzPrevT
                local raptor = Vector3.new()
                if dt > 0 and dt < 0.5 and _reactAlzPrevVel then
                    local accel = (bv - _reactAlzPrevVel) / dt
                    local grav = Vector3.new(0, -28, 0)
                    local netAcc = accel - grav
                    raptor = netAcc * (2.5 + ping * 8)
                end
                _reactAlzPrevVel = bv
                _reactAlzPrevT = now
                local dx = (bp.X + bv.X * lead + raptor.X) - hp.X
                local dy = (bp.Y + bv.Y * lead + raptor.Y) - hp.Y
                local dz = (bp.Z + bv.Z * lead + raptor.Z) - hp.Z
                if dx >= -1.2 and dx <= 1.2 and dz >= -1.2 and dz <= 1.2 and dy >= -0.3 and dy <= 3.0 then
                    if firetouchinterest then
                        firetouchinterest(tpsBall, head, 0)
                        firetouchinterest(tpsBall, head, 1)
                    end
                end

        local function stripFlagName(k)
            return k:gsub("^DFInt",""):gsub("^DFFlag",""):gsub("FString",""):gsub("FLog",""):gsub("^FFlag",""):gsub("^DFint",""):gsub("^FInt","")

        local function writeFlag(rawName, value)
            if not setfflag then
                return
            end

            pcall(function()
                local ck = stripFlagName(rawName)

                if getfflag and getfflag(ck) ~= nil then
                    setfflag(ck, value)
                elseif getfflag and getfflag(rawName) ~= nil then
                    setfflag(rawName, value)
                end
            end)
        end

        local FlagManager = {}
        local fmThread = nil
        local fmRegs = {}
        local fmOrder = {}
        local FM_FLAG_NAMES = {
            tt = "DFIntTargetTimeDelayFacctorTenths",
            interp = "FIntInterpolationMaxDelayMSec",
            s2 = "DFIntS2PhysicsSenderRate",
        }

        local function fmActiveFlags()
            if #fmOrder == 0 then
                return nil
            end
            return fmRegs[fmOrder[#fmOrder]]
        end

        function FlagManager.apply(owner, flags)
            fmRegs[owner] = flags or {}
            for i, o in ipairs(fmOrder) do
                if o == owner then
                    table.remove(fmOrder, i)
                    break
                end
            end
            table.insert(fmOrder, owner)
            fmStartLoop()
        end

        function FlagManager.update(owner, flags)
            if not fmRegs[owner] then
                return
            end
            for k, v in pairs(flags or {}) do
                fmRegs[owner][k] = v
            end
        end

        function FlagManager.stop(owner)
            if not fmRegs[owner] then
                return
            end
            fmRegs[owner] = nil
            for i, o in ipairs(fmOrder) do
                if o == owner then
                    table.remove(fmOrder, i)
                    break
                end
            end
            if #fmOrder == 0 then
                fmStopLoop()
            end
        end

        function FlagManager.hasOwner(owner)
            return fmRegs[owner] ~= nil
        end

        local function fmStartLoop()
            if fmThread then
                return
            end
            fmThread = task.spawn(function()
                while true do
                    local flags = fmActiveFlags()
                    if flags then
                        for name, value in pairs(flags) do
                            local key = FM_FLAG_NAMES[name] or name
                            writeFlag(key, value)
                        end
                    end
                    task.wait(20)
                end
            end)
        end

        local function fmStopLoop()
            if fmThread then
                task.cancel(fmThread)
                fmThread = nil
            end
        end

        local function tmStart()
            tmRestore()

            tmApplyMaxTPS()
            tmExt.apply()

            local cfg = tmCfg()
            tmPrediction = cfg.prediction or tmPrediction
            tmAdaptive = (cfg.adaptive ~= nil) and cfg.adaptive or tmAdaptive
            FlagManager.apply("tmbuzzz", {
                tt = cfg.tt,
                interp = cfg.interp,
                s2 = cfg.s2,
            })

            tmConnection = (RunService.PreSimulation or RunService.Heartbeat):Connect(function()
                if not tmActive then
                    return
                end

                local character = getCharacter()
                local head = character and character:FindFirstChild("Head")
                local tpsBall = getTPSBall()

                if not head or not tpsBall then
                    return
                end

                local now = os.clock()

                if now - tmPingT >= 0.06 then
                    tmPingT = now
                    tmPing = math.clamp(LocalPlayer:GetNetworkPing(), 0.016, 0.20)
                end

                local bp = tpsBall.Position
                local bv = tpsBall.AssemblyLinearVelocity
                local hp = head.Position
                local velMag = bv.Magnitude

                if tmStatus.magnetism then
                    local toHead = hp - bp
                    local dist = toHead.Magnitude
                    if dist > 0.3 and dist < tmStatus.magnetRange then
                        local dir = toHead.Unit
                        local strength = tmStatus.magnetPower * (1 - dist / tmStatus.magnetRange)
                        local magDt = math.clamp(1 / 60, 0, 1)
                        if tpsBall:IsA("BasePart") then
                            local nv = bv + dir * (strength * magDt)
                            local speed = nv.Magnitude
                            local maxSpeed = math.max(velMag, 600)
                            if speed > maxSpeed then
                                nv = nv.Unit * maxSpeed
                            end
                            tpsBall.Velocity = nv
                        end
                    end
                end

                local baseLead = tmPing * 2.8
                if tmPrediction == "LOW" then
                    baseLead = tmPing * 2.0
                elseif tmPrediction == "MED" then
                    baseLead = tmPing * 3.6
                elseif tmPrediction == "HIGH" then
                    baseLead = tmPing * 4.5 + 0.12
                else
                    baseLead = tmPing * 2.2 + 1 / 8
                end

                local dt = now - tmPrevT
                local raptor = Vector3.new()
                if dt > 0 and dt < 0.5 and tmPrevVel then
                    local accel = (bv - tmPrevVel) / dt
                    local grav = Vector3.new(0, -28, 0)
                    local netAcc = accel - grav
                    raptor = netAcc * (4.5 + tmPing * 16)
                end

                tmPrevVel = bv
                tmPrevT = now

                if tmAdaptive then
                    local rv = tpsBall.GetNetworkRotVelocity and tpsBall:GetNetworkRotVelocity()
                    if rv and rv.Magnitude > 0.01 then
                        local axis = rv.Unit
                        raptor = raptor + (axis:Cross(bv) * (0.5 + velMag / 2000))
                    end
                end

                local lead = baseLead + 1 / 60
                if tmAdaptive then
                    lead = lead + math.min(math.max(velMag / 35, 0), 0.60)
                end

                local ax = bp.X + bv.X * lead + raptor.X
                local ay = bp.Y + bv.Y * lead + raptor.Y
                local az = bp.Z + bv.Z * lead + raptor.Z

                local cfg = tmBoxFor()
                local hz = cfg.hz
                local dyMin = cfg.dyMin
                local dyMax = cfg.dyMax
                local hx = cfg.hx
                local dx = ax - hp.X
                local dy = ay - hp.Y
                local dz = az - hp.Z

                local inHead = dx >= -hx and dx <= hx and dz >= -hz and dz <= hz and dy >= dyMin and dy <= dyMax
                local inVortex = tmZones.vortex and not inHead
                    and dx >= -13 and dx <= 13 and dz >= -13 and dz <= 13 and dy >= -1.5 and dy <= 8
                local inAura = not (inHead or inVortex) and (dx * dx + dy * dy + dz * dz) <= (cfg.magR * cfg.magR)

                if not (inHead or inVortex or inAura) then
                    if tmContact then
                        tmContact = false
                        writeFlag("DFIntTargetTimeDelayFacctorTenths", tmCfg().tt)
                        FlagManager.update("tmbuzzz", { tt = tmCfg().tt })
                    end
                else
                    if firetouchinterest then
                        firetouchinterest(tpsBall, head, 0)
                        firetouchinterest(tpsBall, head, 1)
                    end

                    local root = character:FindFirstChild("HumanoidRootPart")
                    if cfg.allParts then
                        local nukeParts = {
                            root,
                            character:FindFirstChild("Torso"),
                            character:FindFirstChild("UpperTorso"),
                            character:FindFirstChild("RightUpperArm"),
                            character:FindFirstChild("LeftUpperArm"),
                            character:FindFirstChild("RightHand"),

                            end

                    elseif root then
                        firetouchinterest(tpsBall, root, 0)
                        firetouchinterest(tpsBall, root, 1)
                    end

                    local rLeg = character:FindFirstChild("RightLowerLeg") or character:FindFirstChild("Right Leg")
                    local lLeg = character:FindFirstChild("LeftLowerLeg") or character:FindFirstChild("Left Leg")
                    if cfg.legs and tmZones.legs and rLeg then firetouchinterest(tpsBall, rLeg, 0); firetouchinterest(tpsBall, rLeg, 1) end
                    if cfg.legs and tmZones.legs and lLeg then firetouchinterest(tpsBall, lLeg, 0); firetouchinterest(tpsBall, lLeg, 1) end

                    if not tmContact then

                        writeFlag("DFIntTargetTimeDelayFacctorTenths", tmCfg().ttContact)

        for i = 1, 8 do
            r2BUF[i] = { t = 0, pos = Vector3.zero, vel = Vector3.zero }

        local function r2Record(ball)
            r2Head = (r2Head % 8) + 1
            local slot = r2BUF[r2Head]
            slot.t = os.clock()
            slot.pos = ball.Position
            slot.vel = ball.AssemblyLinearVelocity

            if r2Len < 8 then
                r2Len = r2Len + 1
            end
        end

        local function r2Frame(offset)
            return r2BUF[((r2Head - offset - 1) % 8) + 1]
        end

        local function r2SmartPredict(ball, ahead)
            if r2Len < 2 then
                return ball.Position + ball.AssemblyLinearVelocity * ahead, ball.AssemblyLinearVelocity
            end

            local s2 = r2Frame(0)
            local s1 = r2Frame(1)
            local dt = s2.t - s1.t

            if dt <= 0 then
                dt = 0.016
            end

            local vpx = (s2.pos.X - s1.pos.X) / dt
            local vpy = (s2.pos.Y - s1.pos.Y) / dt
            local vpz = (s2.pos.Z - s1.pos.Z) / dt
            local vrx = s2.vel.X
            local vry = s2.vel.Y
            local vrz = s2.vel.Z
            local ddx = vrx - vpx
            local ddy = vry - vpy
            local ddz = vrz - vpz
            local kb = math.clamp((ddx * ddx + ddy * ddy + ddz * ddz) / (35 * 35), 0, 1)
            local blx = vpx * (1 - kb) + vrx * kb
            local bly = vpy * (1 - kb) + vry * kb
            local blz = vpz * (1 - kb) + vrz * kb
            local ax, ay, az = 0, 0, 0

            if r2Len >= 3 then
                local s0 = r2Frame(2)
                local dt0 = s1.t - s0.t

                if dt0 > 0 then
                    local v0x = (s1.pos.X - s0.pos.X) / dt0
                    local v0y = (s1.pos.Y - s0.pos.Y) / dt0
                    local v0z = (s1.pos.Z - s0.pos.Z) / dt0
                    local amax = 120
                    ax = math.clamp((vpx - v0x) / dt, -amax, amax)
                    ay = math.clamp((vpy - v0y) / dt, -amax, amax)
                    az = math.clamp((vpz - v0z) / dt, -amax, amax)
                end
            end

            local gravY = -9.8 * ahead * ahead * (1 - kb)
            local t = math.clamp(ahead, 0.016, 0.25)
            local half = 0.5 * t * t

            return Vector3.new(
                s2.pos.X + blx * t + ax * half,
                s2.pos.Y + bly * t + ay * half + gravY,
                s2.pos.Z + blz * t + az * half
            ), Vector3.new(blx, bly, blz)
        end

        local function r2Fire(ball)
            local character = getCharacter()

            if not character then
                return
            end

            local root = character:FindFirstChild("HumanoidRootPart")
            local rLeg = character:FindFirstChild("RightLowerLeg") or character:FindFirstChild("Right Leg")
            local lLeg = character:FindFirstChild("LeftLowerLeg") or character:FindFirstChild("Left Leg")

            if firetouchinterest then
                if root then
                    firetouchinterest(ball, root, 0)
                    firetouchinterest(ball, root, 1)
                end

                if rLeg then
                    firetouchinterest(ball, rLeg, 0)
                    firetouchinterest(ball, rLeg, 1)
                end

                if lLeg then
                    firetouchinterest(ball, lLeg, 0)
                    firetouchinterest(ball, lLeg, 1)
                end
            end
        end

        local GRAV_HALF = -9.8 * 0.5 * (1 / 60) * (1 / 60)

        local function r2FireFrame()
            local character = getCharacter()

            if not character then
                return
            end

            local root = character:FindFirstChild("HumanoidRootPart")
            local tpsBall = getTPSBall()

            if not root or not tpsBall then
                return
            end

            local now = os.clock()

            if now - r2PingT >= 2 then
                r2PingT = now
                r2Ping = math.clamp(LocalPlayer:GetNetworkPing(), 0.016, 0.25)
            end

            local ballPos = tpsBall.Position
            local rootPos = root.Position
            local qdx = ballPos.X - rootPos.X
            local qdz = ballPos.Z - rootPos.Z

            if math.abs(qdx) > 32 or math.abs(qdz) > 32 then
                return
            end

            r2Record(tpsBall)

            local qdy = ballPos.Y - rootPos.Y
            local rawDist = math.sqrt(qdx * qdx + qdy * qdy + qdz * qdz)
            local ahead = r2Ping + 1 / 60
            local predPos, predVel

            if r2Ping >= 0.10 or rawDist >= 7 then
                predPos, predVel = r2SmartPredict(tpsBall, ahead)
            else
                predPos = ballPos
                predVel = tpsBall.AssemblyLinearVelocity
            end

            local farPos = Vector3.new(
                predPos.X + predVel.X * (1 / 60),
                predPos.Y + predVel.Y * (1 / 60) + GRAV_HALF,
                predPos.Z + predVel.Z * (1 / 60)
            )

            local rx = reach2Size.X / 2
            local ry = reach2Size.Y / 2
            local rz = reach2Size.Z / 2
            local dx = predPos.X - rootPos.X
            local dy = predPos.Y - rootPos.Y
            local dz = predPos.Z - rootPos.Z
            local fdx = farPos.X - rootPos.X
            local fdy = farPos.Y - rootPos.Y
            local fdz = farPos.Z - rootPos.Z

            if rawDist < 7 then
                r2Fire(tpsBall)
            end

            if (math.abs(dx) <= rx and math.abs(dy) <= ry and math.abs(dz) <= rz)
                or (math.abs(fdx) <= rx and math.abs(fdy) <= ry and math.abs(fdz) <= rz) then
                r2Fire(tpsBall)
            end

            if not reach2Hidden and reach2Part then
                reach2Part.CFrame = root.CFrame
            end
        end

        local function startMoss()
            stopMoss()

            FlagManager.apply("moss", mossFlags())

            mossState.connection = (RunService.PreSimulation or RunService.Heartbeat):Connect(function()
                if not mossState.active then
                    return
                end

                local character = getCharacter()
                local head = character and character:FindFirstChild("Head")
                local tpsBall = getTPSBall()

                if not head or not tpsBall or not tpsBall.Parent then
                    return
                end

                local now = os.clock()

                if now - mossState.pingT >= 2 then
                    mossState.pingT = now
                    mossState.ping = math.clamp(LocalPlayer:GetNetworkPing(), 0.016, 0.15)
                end

                local ahead = mossState.ping * 1.5 + 1 / 60
                if mossState.prediction == "LOW" then
                    ahead = mossState.ping * 0.65 + 1 / 60
                elseif mossState.prediction == "HIGH" then
                    ahead = mossState.ping * 1.8 + 0.03
                end

                local bp = tpsBall.Position
                local bv = tpsBall.AssemblyLinearVelocity
                local hp = head.Position
                local dx = (bp.X + bv.X * ahead) - hp.X
                local dy = (bp.Y + bv.Y * ahead) - hp.Y
                local dz = (bp.Z + bv.Z * ahead) - hp.Z

                local hx = mossState.depth * mossState.reach
                local hz = mossState.width * mossState.reach
                local hy = mossState.height * mossState.reach
                local scl = mossIntensityCfg().scl

                local buffer = 0.25 * mossState.reach

                if dx >= -hx * scl - buffer and dx <= hx * scl + buffer
                    and dz >= -hz * scl - buffer and dz <= hz * scl + buffer
                    and dy >= -1.0 and dy <= hy * scl + buffer then
                    if firetouchinterest then
                        firetouchinterest(tpsBall, head, 0)
                        firetouchinterest(tpsBall, head, 1)
                    end

                    if not mossState.contact then
                        mossState.contact = true
                        writeFlag("DFIntTargetTimeDelayFacctorTenths", mossIntensityCfg().ttHit)
                        FlagManager.update("moss", { tt = mossIntensityCfg().ttHit })
                    end
                else
                    if mossState.contact then
                        mossState.contact = false
                        writeFlag("DFIntTargetTimeDelayFacctorTenths", mossIntensityCfg().ttIdle)
                        FlagManager.update("moss", { tt = mossIntensityCfg().ttIdle })
                    end
                end
            end)
        end

        MainFrame.Position = UDim2.new(0.5, -290, 0.5, -190)

            if child:IsA("CanvasGroup") then
                child.GroupTransparency = 0

            end
        end

            PageList.Visible = true

            PagesContainer.Visible = true

            NonTabsContainer.Visible = false
        end

        if mScale then
            mScale.Scale = 1

        if not uiAnimEnabled() then
            return

        fadeGroup(MainFrame, 1, true, 0.001)

        if mScale then
            mScale.Scale = 0.93
        end
        MainFrame.Position = UDim2.new(0.5, -290, 0.53, -190)
        task.wait(0.04)
        restoreFade(MainFrame)
        if mScale then
            tweenUI(mScale, 0.4, Enum.EasingStyle.Quint, Enum.EasingDirection.Out, { Scale = compactScale() })
        end
        tweenUI(MainFrame, 0.45, Enum.EasingStyle.Quint, Enum.EasingDirection.Out, {
            Position = UDim2.new(0.5, -290, 0.5, -190),
        })
        task.wait(0.5)
        MainFrame.BackgroundTransparency = 0
        for _, child in ipairs(PagesContainer:GetChildren()) do
            if child:IsA("CanvasGroup") then
                child.GroupTransparency = 0
                child.Visible = true
            end
        end
        for _, obj in ipairs(MainFrame:GetDescendants()) do
            if obj:IsA("Frame") or obj:IsA("ScrollingFrame") then
                obj.BackgroundTransparency = 0
            elseif obj:IsA("TextLabel") or obj:IsA("TextButton") then
                obj.TextTransparency = 0
            elseif obj:IsA("ImageLabel") then
                obj.ImageTransparency = 0
            elseif obj:IsA("UIStroke") then
                obj.Transparency = 0
            elseif obj:IsA("CanvasGroup") then
                obj.GroupTransparency = 0
            end
        end
        if mScale then
            mScale.Scale = compactScale()
        end
    end

    local function validateGUI()
        if not MainFrame or not MainFrame.Parent then return end
        if not PagesContainer or not PagesContainer.Parent then return end
        if not PageList then return end

        local pageCount = 0
        for _, child in ipairs(PagesContainer:GetChildren()) do
            if child:IsA("CanvasGroup") then
                pageCount = pageCount + 1
            end
        end

        if pageCount == 0 then return end

        local firstPage = PagesContainer:FindFirstChildWhichIsA("CanvasGroup")
        if firstPage then
            firstPage.Visible = true
            firstPage.GroupTransparency = 0
        end

        PageList.Visible = true
        PagesContainer.Visible = true
        NonTabsContainer.Visible = false
        MainFrame.BackgroundTransparency = 0
        local mainScale = MainFrame:FindFirstChildOfClass("UIScale")
        if mainScale then
            mainScale.Scale = compactScale()
        end
        for _, obj in ipairs(MainFrame:GetDescendants()) do
            if obj:IsA("Frame") or obj:IsA("ScrollingFrame") then
                obj.BackgroundTransparency = 0
            elseif obj:IsA("TextLabel") or obj:IsA("TextButton") then
                obj.TextTransparency = 0
            elseif obj:IsA("ImageLabel") then
                obj.ImageTransparency = 0
            elseif obj:IsA("UIStroke") then
                obj.Transparency = 0
            elseif obj:IsA("CanvasGroup") then
                obj.GroupTransparency = 0
            end
        end
    end

     validateGUI()
     pcall(playOpening)
     validateGUI()

end
